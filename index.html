💻 Source Code & Firmware

하드웨어를 제어하는 핵심 펌웨어 아카이브입니다.

motor_control.ino 
// Motor Control Firmware for Autonomous Rover
const int ENA = 9; // PWM 속도 제어 핀
const int IN1 = 8; // 모터 회전 방향 핀 1
const int IN2 = 7; // 모터 회전 방향 핀 2

void setup() {
  pinMode(ENA, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
}

void loop() {
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
  analogWrite(ENA, 200); // 0~255 값으로 속도 제어
}
